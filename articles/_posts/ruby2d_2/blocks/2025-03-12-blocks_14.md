### refrect に左・右を追加する

app/block.rb（`refrect` を更新）

```ruby
def refrect(ball, blocks)
  top_refrect(ball, blocks)    ||
  bottom_refrect(ball, blocks) ||
  right_refrect(ball, blocks)  ||
  left_refrect(ball, blocks)
end
```

### right_refrect と left_refrect を追加する

app/block.rb（`private` の中に追加）

```ruby
def right_refrect(ball, blocks)
  contain = ball.contains?(self.x + self.width + ball.radius, ball.y)
  range   = ball.y >= self.y && ball.y <= self.y + self.height
  if contain && range
    ball.x_flag = true
    self.remove
    blocks.delete(self)
  end
end

def left_refrect(ball, blocks)
  contain = ball.contains?(self.x - ball.radius, ball.y)
  range   = ball.y >= self.y && ball.y <= self.y + self.height
  if contain && range
    ball.x_flag = false
    self.remove
    blocks.delete(self)
  end
end
```

上・下と同じ構造で、今度は y の範囲チェックと x のフラグ変更になります。

### ここまでのコード

app/block.rb

```ruby
class Block < Rectangle
  BLOCK_COLORS = %w(
    red
    orange
    yellow
    green
    blue
    fuchsia
    purple
  )

  def initialize
    super(width: 80, height: 25)
    self.x = self.width / 2
    self.y = self.height * 2
  end

  def self.set_blocks
    BLOCK_COLORS.map.with_index(0) do |color, r|
      9.times.map do |c|
        block = self.new
        block.color = color
        block.x += block.width * c
        block.y += block.height * r
        block.width  -= 1
        block.height -= 1
        block
      end
    end.flatten
  end

  def self.broken_by(ball, blocks)
    blocks.each do |block|
      block.refrect(ball, blocks)
    end
  end

  def refrect(ball, blocks)
    top_refrect(ball, blocks)    ||
    bottom_refrect(ball, blocks) ||
    right_refrect(ball, blocks)  ||
    left_refrect(ball, blocks)
  end

  private

  def top_refrect(ball, blocks)
    contain = ball.contains?(ball.x, self.y - ball.radius)
    range   = ball.x >= self.x && ball.x <= self.x + self.width
    if contain && range
      ball.y_flag = false
      self.remove
      blocks.delete(self)
    end
  end

  def bottom_refrect(ball, blocks)
    contain = ball.contains?(ball.x, self.y + self.height + ball.radius)
    range   = ball.x >= self.x && ball.x <= self.x + self.width
    if contain && range
      ball.y_flag = true
      self.remove
      blocks.delete(self)
    end
  end

  def right_refrect(ball, blocks)
    contain = ball.contains?(self.x + self.width + ball.radius, ball.y)
    range   = ball.y >= self.y && ball.y <= self.y + self.height
    if contain && range
      ball.x_flag = true
      self.remove
      blocks.delete(self)
    end
  end

  def left_refrect(ball, blocks)
    contain = ball.contains?(self.x - ball.radius, ball.y)
    range   = ball.y >= self.y && ball.y <= self.y + self.height
    if contain && range
      ball.x_flag = false
      self.remove
      blocks.delete(self)
    end
  end
end
```

main.rb（変更なし）

```ruby
require "ruby2d"
require_relative "app/bar"
require_relative "app/ball"
require_relative "app/block"

set width: 800, height: 600

bar    = Bar.new
ball   = Ball.new
blocks = Block.set_blocks

on :key do |event|
  bar.move(event)
end

on :key_down do |event|
  if event.key == "space" && !ball.is_move
    ball.is_move = true
  end
end

update do
  ball.move(bar)
  Block.broken_by(ball, blocks)
end

show
```
