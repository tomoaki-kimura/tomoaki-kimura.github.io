### Burnクラスを作る

`app/models/burn.rb` を新規作成しましょう。

app/models/burn.rb

```ruby
class Burn < Sprite
  attr_accessor :tick

  def initialize(x, y)
    super("./app/images/burn.png",
      width: 30,
      height: 30,
      time: 50,
      animations: {
        bomb: [
          { width: 0,  height: 0,  x: 0,  y: 0  },
          { width: 15, height: 15, x: 0,  y: 0  },
          { width: 15, height: 15, x: 15, y: 0  },
          { width: 15, height: 15, x: 0,  y: 15 },
          { width: 15, height: 15, x: 15, y: 15 }
        ]
      }
    )
    self.x = x - self.width / 2
    self.y = y - self.height / 2
    self.play(animation: :bomb)
    self.tick = 0
  end

  def self.selected_delete(burns)
    burns.each do |burn|
      if burn.tick > 250
        burn.remove
        burns.delete(burn)
      else
        burn.tick += 1
      end
    end
  end
end
```

### Bullet に hit メソッドを追加する

敵に当たったとき、爆発生成・アイテムドロップ・敵と弾の削除をまとめて行います。

`self.move` にも `items` を渡せるよう引数を更新します。

app/models/bullet.rb

```ruby
class Bullet < Circle
  ANGLE_OFFSET = [0, -15, 15, -30, 30]
  attr_accessor :speed, :angle

  def initialize
    super(radius: 3)
    self.speed = 15
    self.angle = 0
  end

  def self.set(bullets, fighter)
    bullet_size = fighter.level * 2 - 1
    bullet_size.times do |i|
      bullet = self.new
      bullet.x = fighter.x + (fighter.width / 2)
      bullet.y = fighter.y
      bullet.angle = ANGLE_OFFSET[i]
      bullets << bullet
    end
  end

  def self.move(bullets, enemies, burns, items)
    bullets.each do |bullet|
      r = (bullet.angle + 90) * PI / 180
      bullet.x += cos(r) * bullet.speed
      bullet.y -= sin(r) * bullet.speed
      bullet.hit(enemies, bullets, burns, items)
    end
  end

  def hit(enemies, bullets, burns, items)
    enemies.each do |enemy|
      if enemy.contains?(self.x, self.y)
        burns << Burn.new(enemy.x + enemy.width / 2, enemy.y + enemy.height / 2)
        ItemPow.set([enemy.x, enemy.y], items) if rand(0..3) == 0
        enemy.remove
        enemies.delete(enemy)
        self.remove
        bullets.delete(self)
        break
      end
    end
  end
end
```

`rand(0..3) == 0` で25%の確率でアイテムをドロップします。

### app/main.rb を更新する

`Bullet.move` に引数を追加し、`Burn` の後処理と `burns` 配列を追加します。また確認用の `I` キーを削除します。

app/main.rb

```ruby
require "./config"

@status = Status.new
fighter = Fighter.new
game_title = GameTitle.new
bullets = []
stars = []
items = []
enemies = []
burns = []
tick = 0

update do
  Star.set(stars) if tick % 10 == 0
  Star.move(stars)

  case @status.game_status
  when :title
  when :start
    fighter.opening_action
    Bullet.move(bullets, enemies, burns, items)
    items.each { |item| item.move }
    items.each { |item| item.touch(fighter, items) }
    enemies << Enemy1.new if tick % 20 == 0 && rand(0..2) == 0
    enemies.each { |enemy| enemy.move(enemies) }
    Burn.selected_delete(burns)
  end
  tick += 1
end

on :key do |event|
  key = event.key
  case fighter.status
  when :ready
    fighter.move(key)
  end
end

on :key_down do |event|
  key = event.key
  case key
  when "space"
    if @status.game_status == :title
      game_title.remove
      fighter.status = :clear
      @status.to_start
      fighter.add
    end
  when "j"
    if fighter.status == :ready
      Bullet.set(bullets, fighter)
    end
  end
end

on :key_up do |event|
  fighter.nutral(event.key)
end

show
```

### 結果

弾が敵に当たると爆発アニメーションが再生され、25%の確率でアイテムがドロップされる。

#### *ここで試したいこと

- `rand(0..3) == 0` の数字を変えてドロップ率を変えてみよう（`rand(0..1)` なら50%）。
- `Burn.selected_delete` の `250` を変えて爆発の表示時間を変えてみよう。
