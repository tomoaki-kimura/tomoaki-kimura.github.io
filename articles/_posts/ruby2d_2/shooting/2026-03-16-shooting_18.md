### GameOverクラスを作る

`app/messages/game_over.rb` を新規作成しましょう。

app/messages/game_over.rb

```ruby
class GameOver
  attr_accessor :title, :message

  def initialize
    self.title = Text.new("GAME OVER", size: 60)
    self.title.x = (Window.width - self.title.width) / 2
    self.title.y = 300
    self.message = Text.new("press key 'R' to restart", size: 30)
    self.message.x = (Window.width - self.message.width) / 2
    self.message.y = 420
  end

  def add
    self.title.add
    self.message.add
  end

  def remove
    self.title.remove
    self.message.remove
  end
end
```

### Status に to_game_over を追加する

app/status.rb

```ruby
class Status
  attr_reader :game_status

  def initialize
    to_title
  end

  def to_title
    @game_status = :title
  end

  def to_start
    @game_status = :start
  end

  def to_game_over
    @game_status = :game_over
  end
end
```

### config.rb に require を追加する

config.rb

```ruby
require "./app/messages/game_title"
```

### app/main.rb にゲームオーバーとリスタート処理を追加する

app/main.rb

```ruby
require "./config"

@status = Status.new
fighter = Fighter.new
game_title = GameTitle.new
game_over = GameOver.new
bullets = []
stars = []
items = []
enemies = []
burns = []
tick = 0

update do
  Star.set(stars) if tick % 10 == 0
  Star.move(stars)

  case @status.game_status
  when :title
  when :start
    fighter.opening_action
    fighter.update_invincible
    Bullet.move(bullets, enemies, burns, items)
    items.each { |item| item.move }
    items.each { |item| item.touch(fighter, items) }
    enemies << Enemy1.new if tick % 20 == 0 && rand(0..2) == 0
    enemies.each do |enemy|
      enemy.move(enemies)
      fighter.bloken_by(enemy, enemies)
    end
    Burn.selected_delete(burns)
    if fighter.life <= 0
      fighter.remove
      @status.to_game_over
      game_over.add
    end
  end
  tick += 1
end

on :key do |event|
  key = event.key
  case fighter.status
  when :ready
    fighter.move(key)
  end
end

on :key_down do |event|
  key = event.key
  case key
  when "space"
    if @status.game_status == :title
      game_title.remove
      fighter.status = :clear
      @status.to_start
      fighter.add
    end
  when "j"
    if fighter.status == :ready
      Bullet.set(bullets, fighter)
    end
  when "r"
    if @status.game_status == :game_over
      game_over.remove
      enemies.each(&:remove)
      enemies.clear
      bullets.each(&:remove)
      bullets.clear
      items.each(&:remove)
      items.clear
      burns.each(&:remove)
      burns.clear
      fighter.life = 3
      fighter.level = 1
      fighter.pow = 0
      fighter.invincible = false
      fighter.reload_life
      fighter.status = :clear
      fighter.add
      @status.to_start
    end
  end
end

on :key_up do |event|
  fighter.nutral(event.key)
end

show
```

`reload_life` は `private` なので `fighter.reload_life` では呼べません。`public` に移すか、`Fighter` に `reset` メソッドを作ってまとめるとよいでしょう。

リスタート用にまとめたメソッドを Fighter に追加します。

app/models/fighter.rb（public に reset を追加）

```ruby
def reset
  self.life = 3
  self.level = 1
  self.pow = 0
  self.invincible = false
  self.check_point = 0
  reload_life
end
```

main.rb の `r` キー処理はこうシンプルになります。

```ruby
when "r"
  if @status.game_status == :game_over
    game_over.remove
    [enemies, bullets, items, burns].each { |arr| arr.each(&:remove); arr.clear }
    fighter.reset
    fighter.status = :clear
    fighter.add
    @status.to_start
  end
```

### ファイルの実行

```bash
shooting $ bundle exec rake
```

### 結果

ライフが0になるとゲームオーバー画面が表示され、Rキーで最初からプレイできる。

- ライフの初期値を変えてみよう。
