### 当たり判定エリアを作る

Spriteは四角形なので、中心付近の4点で当たり判定をとります。Fighter クラスに `hit_area` メソッドを追加します。

app/models/fighter.rb（メソッドを追加）

```ruby
def hit_area
  offset = 10
  offset_map = [offset * -1, offset]
  area = []
  offset_map.each do |y|
    offset_map.each do |x|
      area << [
        self.x + self.width / 2 + x,
        self.y + self.height / 2 + y
      ]
    end
  end
  area
end
```

自機の中心から上下左右に `offset` ずらした4点の座標を配列で返します。

### ItemPow に touch メソッドを追加する

app/models/item_pow.rb

```ruby
class ItemPow < Sprite
  attr_accessor :speed

  def initialize
    super("./app/images/item_pow.png",
      width: 30,
      height: 30
    )
    self.speed = 3
  end

  def move
    self.y += self.speed
  end

  def touch(fighter, items)
    fighter.hit_area.each do |point|
      if self.contains?(*point)
        self.remove
        items.delete(self)
        fighter.power_up
        break
      end
    end
  end

  def self.set(position, items)
    item = self.new
    item.x = position.first
    item.y = position.last
    items << item
  end
end
```

### Fighter に power_up / power_down メソッドを追加する

app/models/fighter.rb

```ruby
def power_up
  if self.level < 3
    if self.pow < 2
      self.pow += 1
    else
      self.pow = 0
      self.level += 1
    end
  end
end

def power_down
  self.level -= 1 if self.level > 0
  self.pow = 0
end
```

アイテムを3個取るたびにレベルが1上がります（最大3）。

app/models/fighter.rb（hit_area・power_up・power_down を追加した全文）

```ruby
class Fighter < Sprite
  attr_accessor :speed, :tick, :status, :pow, :level, :life, :life_gauge

  def initialize
    super("app/images/fighter.png")
    self.width = 30
    self.height = 30
    self.clip_width = 15
    self.clip_height = 15
    self.speed = 3.0
    self.remove
    self.tick = 0
    self.status = nil
    self.pow = 0
    self.level = 1
    self.life = 3
    self.life_gauge = Text.new("Life: #{self.life}", x: 20, y: 15)
  end

  def move(key)
    right_move if key == "d"
    left_move  if key == "a"
    self.y -= self.speed if key == "w"
    self.y += self.speed if key == "s"
    window_limit
  end

  def nutral(key)
    center_image if key == "d" || key == "a"
  end

  def opening_action
    case self.status
    when :clear
      self.x = (Window.width - self.width) / 2
      self.y = Window.height - self.height
      self.status = :starting
    when :starting
      self.y -= 1
      self.color = "random"
      if self.y < Window.height / 4 * 3
        self.status = :ready
        self.color = "white"
      end
    end
  end

  def hit_area
    offset = 10
    offset_map = [offset * -1, offset]
    area = []
    offset_map.each do |y|
      offset_map.each do |x|
        area << [
          self.x + self.width / 2 + x,
          self.y + self.height / 2 + y
        ]
      end
    end
    area
  end

  def power_up
    if self.level < 3
      if self.pow < 2
        self.pow += 1
      else
        self.pow = 0
        self.level += 1
      end
    end
  end

  def power_down
    self.level -= 1 if self.level > 0
    self.pow = 0
  end

  private

  def right_move
    self.x += self.speed
    self.clip_x = 15
    self.flip_sprite(nil)
  end

  def left_move
    self.x -= self.speed
    self.clip_x = 15
    self.flip_sprite(:horizontal)
  end

  def center_image
    self.clip_x = 0
    self.flip_sprite(nil)
  end

  def window_limit
    self.y = 0 if self.y < 0
    self.y = Window.height - self.height if self.y > Window.height - self.height
    self.x = 0 if self.x < 0
    self.x = Window.width - self.width if self.x > Window.width - self.width
  end
end
```
