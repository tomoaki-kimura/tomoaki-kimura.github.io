### ここまでのコード

main.rb
```
require "ruby2d"

head = Circle.new(radius: 10)
head.x = Window.width / 2
head.y = Window.height / 2
speed = 1
x_flag = true
y_flag = false

on :key_down do |event|
  # 右上に移動の時
  if x_flag && !y_flag
    case event.key
    when "right"
      y_flag = true
    when "left"
      x_flag = false
    end
  # 左上に移動の時
  elsif !x_flag && !y_flag
    case event.key
    when "right"
      x_flag = true
    when "left"
      y_flag = true
    end
  # 右下に移動の時
  elsif x_flag && y_flag
    case event.key
    when "right"
      x_flag = false
    when "left"
      y_flag = false
    end
  # 左下に移動の時
  elsif !x_flag && y_flag
    case event.key
    when "right"
      y_flag = false
    when "left"
      x_flag = true
    end
  end
end

update do
  if x_flag
    head.x += speed
  else
    head.x -= speed
  end
  if y_flag
    head.y += speed
  else
    head.y -= speed
  end
end

show
```
